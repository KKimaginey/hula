// ------------------------------ Utility: create flat SVG icon per card (no gradients) ------------------------------

function makeIconSVG(title, color){

  const initials = title.split(' ')[0].slice(0,2).replace(/[^A-Za-zÀ-ſ0-9ʻʻ]/g,'').toUpperCase();

  const bg = color || '#e6f7f6';

  const fg = '#063636';

  const svg = `<svg xmlns='http://www.w3.org/2000/svg' width='120' height='120' viewBox='0 0 120 120'><rect width='120' height='120' fill='${bg}' rx='12'/><text x='50%' y='52%' dominant-baseline='middle' text-anchor='middle' font-family='Inter, Noto Sans KR, sans-serif' font-size='36' fill='${fg}'>${escapeHtml(initials)}</text></svg>`;

  return 'data:image/svg+xml;utf8,' + encodeURIComponent(svg);

}

function escapeHtml(s){return (s||'').toString().replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')}

// ------------------------------ Render cards ------------------------------

const gridEl = document.getElementById('cardGrid');

const cardCountEl = document.getElementById('cardCount');

const searchEl = document.getElementById('search');

function renderCards(filter=''){

  gridEl.innerHTML='';

  const query = filter.trim().toLowerCase();

  const list = cardDeck.filter(c=>{ return !query || (c.title + ' ' + c.meaning).toLowerCase().includes(query); });

  cardCountEl.textContent = list.length;

  list.forEach(c=>{

    const outer = document.createElement('div'); outer.className='card'; outer.dataset.id = c.id;

    const inner = document.createElement('div'); inner.className='card-inner';

    const front = document.createElement('div'); front.className='face front';

    const img = document.createElement('img'); img.className='icon'; img.src = c.icon; img.alt = c.title;

    const h = document.createElement('h4'); h.textContent = c.title;

    front.appendChild(img); front.appendChild(h);

    const back = document.createElement('div'); back.className='face back';

    const bh = document.createElement('h4'); bh.textContent = c.title;

    const bp = document.createElement('p'); bp.textContent = c.meaning;

    back.appendChild(bh); back.appendChild(bp);

    inner.appendChild(front); inner.appendChild(back); outer.appendChild(inner);

    outer.addEventListener('click', ()=>{

      outer.classList.toggle('is-flipped');

      // also open modal with full content

      openModal(c);

    });

    gridEl.appendChild(outer);

  });

}

// ------------------------------ Shuffle / Export ------------------------------

function shuffleDeck(){ for(let i=cardDeck.length-1;i>0;i--){ const j = Math.floor(Math.random()*(i+1)); [cardDeck[i], cardDeck[j]] = [cardDeck[j], cardDeck[i]]; } renderCards(searchEl.value); }

function exportJSON(){ const a = document.createElement('a'); a.href = 'data:application/json;charset=utf-8,' + encodeURIComponent(JSON.stringify(cardDeck, null, 2)); a.download = 'maginey_mana_cards.json'; a.click(); }


// ------------------------------ Links (Profile -> Practice) ------------------------------

function loadSavedLinks(){ const arr = JSON.parse(localStorage.getItem('mag_links')||'[]'); const el = document.getElementById('savedLinks'); el.innerHTML = arr.length? arr.map((it,idx)=>`<div style="display:flex;justify-content:space-between;align-items:center;padding:8px;border-radius:8px;background:#fbfbfb;margin-bottom:8px"><div><strong>${escapeHtml(it.title)}</strong><div class='small'>${escapeHtml(it.url)}</div></div><div><button onclick="window.open('${encodeURI(''+(arr[idx].url))}','_blank')">Open</button> <button onclick="delLink(${idx})">Del</button></div></div>`).join('') : '<div class="small">저장된 링크가 없습니다.</div>' }

function delLink(i){ const a=JSON.parse(localStorage.getItem('mag_links')||'[]'); a.splice(i,1); localStorage.setItem('mag_links',JSON.stringify(a)); loadSavedLinks(); renderYtLinks(); }

function addLink(){ const title = document.getElementById('linkTitle').value.trim(); const url = document.getElementById('linkUrl').value.trim(); if(!title||!url){ alert('제목과 URL을 입력하세요'); return } const arr = JSON.parse(localStorage.getItem('mag_links')||'[]'); arr.push({title,url}); localStorage.setItem('mag_links',JSON.stringify(arr)); document.getElementById('linkTitle').value=''; document.getElementById('linkUrl').value=''; loadSavedLinks(); renderYtLinks(); }

function renderYtLinks(){ const arr = JSON.parse(localStorage.getItem('mag_links')||'[]'); const el = document.getElementById('ytLinks'); el.innerHTML = arr.length? arr.map(it=>`<div style="display:flex;justify-content:space-between;align-items:center;padding:8px;border-radius:8px;background:#fff;margin-bottom:8px"><div><strong>${escapeHtml(it.title)}</strong><div class='small'>${escapeHtml(it.url)}</div></div><div><button onclick="window.open('${encodeURI(it.url)}','_blank')">Open</button></div></div>`).join('') : '<div class="small">저장된 라이브 링크가 없습니다.</div>' }
